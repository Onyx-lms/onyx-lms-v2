from statistics import mean
from sensors import read_batches   # stdlib only — the sandbox has no network

THRESHOLD_C = 40.0


def summarise(readings):
    """Collapse one sensor's readings into a single row of the report."""
    values = [r.celsius for r in readings]
    return {
        "sensor": readings[0].sensor_id,
        "n": len(values),
        "mean": round(mean(values), 1),
        "peak": max(values),
        "over": sum(1 for v in values if v > THRESHOLD_C),
    }


def main():
    batches = read_batches("data/readings.csv")
    hot = 0

    for sensor_id in sorted(batches):
        row = summarise(batches[sensor_id])
        hot += 1 if row["over"] else 0
        flag = "OVER" if row["over"] else "ok"
        print(f"{row['sensor']:<9} n={row['n']:<4} "
              f"mean={row['mean']:>5} peak={row['peak']:>5}  {flag}")

    print(f"\n{len(batches)} sensors, {hot} above {THRESHOLD_C}C")


if __name__ == "__main__":
    main()
